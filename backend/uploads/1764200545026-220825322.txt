    <script>
        const slides = document.querySelectorAll('.slide');
        let currentSlide = 0;
        const apiKey = ""; // API key is provided at runtime

        function showSlide(index) {
            slides.forEach((slide, i) => {
                slide.classList.remove('active');
                if (i === index) {
                    slide.classList.add('active');
                    // Reset animation for content
                    const contentElements = slide.querySelectorAll('.slide-content > *');
                    contentElements.forEach(el => {
                        el.style.animation = 'none';
                        el.offsetHeight; /* trigger reflow */
                        el.style.animation = '';
                    });
                }
            });
            updateNavigation();
        }

        function navigateSlide(direction) {
            currentSlide = (currentSlide + direction + slides.length) % slides.length;
            showSlide(currentSlide);
        }

        function updateNavigation() {
            document.getElementById('prevBtn').style.display = currentSlide === 0 ? 'none' : 'block';
            document.getElementById('nextBtn').style.display = currentSlide === slides.length - 1 ? 'none' : 'block';
        }

        function closeModal() {
            document.getElementById('llmModal').style.display = 'none';
        }

        async function getLlmContent(strategyName, contentType) {
            document.getElementById('llmLoading').style.display = 'flex';
            
            let userQuery = '';
            let systemPrompt = '';
            let useSearch = false;

            if (contentType === 'أمثلة واقعية') {
                document.getElementById('modalTitle').textContent = `أمثلة واقعية لـ: ${strategyName}`;
                userQuery = `قدم 3 أمثلة حديثة وواقعية وواضحة لـ: "${strategyName}" في عالم الأعمال. يجب أن يكون كل مثال في سطر منفصل، وأن يكون موجزاً ومركّزاً على اسم الاستراتيجية.`;
                systemPrompt = "أنت مساعد خبير في الإدارة الاستراتيجية. مهمتك هي توليد أمثلة تطبيقية واقعية ومقنعة للاستراتيجيات المطلوبة باللغة العربية. يجب أن تكون الأمثلة من الشركات المعروفة أو الصناعات الحديثة.";
                useSearch = true;
            } else if (contentType === 'أسئلة مناقشة') {
                document.getElementById('modalTitle').textContent = `أسئلة مناقشة حول: ${strategyName}`;
                userQuery = `قم بتوليد 3 أسئلة مناقشة عميقة ومحفزة للطلاب حول استراتيجية: "${strategyName}". ركز على أسئلة تتطلب تحليلاً أو تقييماً نقدياً وليس مجرد تعريفات. يجب أن يكون كل سؤال في سطر منفصل.`;
                systemPrompt = "أنت مصمم مناهج أكاديمي. مهمتك هي توليد أسئلة مناقشة نقدية وعالية الجودة في مجال الإدارة الاستراتيجية.";
                useSearch = false; // Discussion questions do not need grounding
            }
            
            document.getElementById('llmOutput').textContent = 'جاري التوليد...';

            const apiUrl = `https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash-preview-05-20:generateContent?key=${apiKey}`;

            const payload = {
                contents: [{ parts: [{ text: userQuery }] }],
                systemInstruction: { parts: [{ text: systemPrompt }] },
            };

            if (useSearch) {
                payload.tools = [{ "google_search": {} }];
            }

            const maxRetries = 3;
            let currentRetry = 0;

            while (currentRetry < maxRetries) {
                try {
                    const response = await fetch(apiUrl, {
                        method: 'POST',
                        headers: { 'Content-Type': 'application/json' },
                        body: JSON.stringify(payload)
                    });

                    if (!response.ok) {
                        throw new Error(`HTTP error! status: ${response.status}`);
                    }

                    const result = await response.json();
                    const text = result.candidates?.[0]?.content?.parts?.[0]?.text || "تعذر توليد المحتوى حالياً.";
                    
                    document.getElementById('llmOutput').textContent = text;
                    document.getElementById('llmModal').style.display = 'block';
                    document.getElementById('llmLoading').style.display = 'none';
                    return; // Exit loop on success

                } catch (error) {
                    console.error("Fetch error:", error);
                    currentRetry++;
                    if (currentRetry < maxRetries) {
                        const delay = Math.pow(2, currentRetry) * 1000;
                        await new Promise(resolve => setTimeout(resolve, delay));
                    }
                }
            }
            
            // If all retries fail
            document.getElementById('llmOutput').textContent = "فشل في الاتصال بخدمة توليد المحتوى بعد عدة محاولات.";
            document.getElementById('llmModal').style.display = 'block';
            document.getElementById('llmLoading').style.display = 'none';
        }

        // Initialize presentation
        document.addEventListener('DOMContentLoaded', () => {
            slides.forEach((slide, i) => {
                if (i !== 0) {
                    slide.style.position = 'absolute';
                    slide.style.top = 0;
                    slide.style.left = 0;
                }
            });
            showSlide(currentSlide);
        });

        // Keyboard Navigation (Arrow Keys)
        document.addEventListener('keydown', (e) => {
            if (e.key === 'ArrowRight' || e.key === ' ') {
                if (currentSlide < slides.length - 1) {
                    navigateSlide(1);
                }
            } else if (e.key === 'ArrowLeft') {
                if (currentSlide > 0) {
                    navigateSlide(-1);
                }
            }
        });
    </script>
